35/// MergeAppendAction is a transaction action similar to fast append except that it will merge manifests
36/// based on the target size.
37pub struct MergeAppendAction {
38    target_size_bytes: u32,
39    min_count_to_merge: u32,
40    check_duplicate: bool,
41    merge_enabled: bool,
42    // below are properties used to create SnapshotProducer when commit
43    commit_uuid: Option<Uuid>,
44    key_metadata: Option<Vec<u8>>,
45    snapshot_properties: HashMap<String, String>,
46    added_data_files: Vec<DataFile>,
47}
48
49/// Target size of manifest file when merging manifests.
50pub const MANIFEST_TARGET_SIZE_BYTES: &str = "commit.manifest.target-size-bytes";
51const MANIFEST_TARGET_SIZE_BYTES_DEFAULT: u32 = 8 * 1024 * 1024; // 8 MB
52/// Minimum number of manifests to merge.
53pub const MANIFEST_MIN_MERGE_COUNT: &str = "commit.manifest.min-count-to-merge";
54const MANIFEST_MIN_MERGE_COUNT_DEFAULT: u32 = 100;
55/// Whether allow to merge manifests.
56pub const MANIFEST_MERGE_ENABLED: &str = "commit.manifest-merge.enabled";
57const MANIFEST_MERGE_ENABLED_DEFAULT: bool = false;
58
59impl MergeAppendAction {
60    pub(crate) fn new(table: &Table) -> Result<Self> {
61        let target_size_bytes: u32 = parse_property(
62            table.metadata().properties(),
63            MANIFEST_TARGET_SIZE_BYTES,
64            MANIFEST_TARGET_SIZE_BYTES_DEFAULT,
65        )?;
66        let min_count_to_merge: u32 = parse_property(
67            table.metadata().properties(),
68            MANIFEST_MIN_MERGE_COUNT,
69            MANIFEST_MIN_MERGE_COUNT_DEFAULT,
70        )?;
71        let merge_enabled = parse_property(
72            table.metadata().properties(),
73            MANIFEST_MERGE_ENABLED,
74            MANIFEST_MERGE_ENABLED_DEFAULT,
75        )?;
76        Ok(Self {
77            check_duplicate: true,
78            target_size_bytes,
79            min_count_to_merge,
80            merge_enabled,
81            commit_uuid: None,
82            key_metadata: None,
83            snapshot_properties: HashMap::default(),
84            added_data_files: vec![],
85        })
86    }
87
88    pub fn with_check_duplicate(mut self, v: bool) -> Self {
89        self.check_duplicate = v;
90        self
91    }
92
93    pub fn set_commit_uuid(mut self, commit_uuid: Uuid) -> Self {
94        self.commit_uuid = Some(commit_uuid);
95        self
96    }
97
98    pub fn set_key_metadata(mut self, key_metadata: Vec<u8>) -> Self {
99        self.key_metadata = Some(key_metadata);
100        self
101    }
102
103    pub fn set_snapshot_properties(mut self, snapshot_properties: HashMap<String, String>) -> Self {
104        self.snapshot_properties = snapshot_properties;
105        self
106    }
107
108    /// Add data files to the snapshot.
109    pub fn add_data_files(
110        &mut self,
111        data_files: impl IntoIterator<Item = DataFile>,
112    ) -> Result<&mut Self> {
113        self.added_data_files.extend(data_files);
114        Ok(self)
115    }
116}
117
118#[async_trait]
119impl TransactionAction for MergeAppendAction {
120    async fn commit(self: Arc<Self>, table: &Table) -> Result<ActionCommit> {
121        let snapshot_producer = SnapshotProducer::new(
122            table,
123            self.commit_uuid.unwrap_or_else(Uuid::now_v7),
124            self.key_metadata.clone(),
125            self.snapshot_properties.clone(),
126            self.added_data_files.clone(),
127        );
128
129        // validate added files
130        snapshot_producer.validate_added_data_files(&self.added_data_files)?;
131
132        // Checks duplicate files
133        if self.check_duplicate {
134            snapshot_producer
135                .validate_duplicate_files(&self.added_data_files)
136                .await?;
137        }
138
139        if self.merge_enabled {
140            snapshot_producer
141                .commit(AppendOperation, MergeManifsetProcess {
142                    target_size_bytes: self.target_size_bytes,
143                    min_count_to_merge: self.min_count_to_merge,
144                })
145                .await
146        } else {
147            snapshot_producer
148                .commit(AppendOperation, DefaultManifestProcess)
149                .await
150        }
151    }
152}

154struct MergeManifestManager {
155    target_size_bytes: u32,
156    min_count_to_merge: u32,
157    content: ManifestContentType,
158}
159
160impl MergeManifestManager {
161    pub fn new(
162        target_size_bytes: u32,
163        min_count_to_merge: u32,
164        content: ManifestContentType,
165    ) -> Self {
166        Self {
167            target_size_bytes,
168            min_count_to_merge,
169            content,
170        }
171    }
172
173    fn group_by_spec(&self, manifests: Vec<ManifestFile>) -> BTreeMap<i32, Vec<ManifestFile>> {
174        let mut grouped_manifests = BTreeMap::new();
175        for manifest in manifests {
176            grouped_manifests
177                .entry(manifest.partition_spec_id)
178                .or_insert_with(Vec::new)
179                .push(manifest);
180        }
181        grouped_manifests
182    }
183
184    async fn merge_bin(
185        &self,
186        snapshot_id: i64,
187        file_io: FileIO,
188        manifest_bin: Vec<ManifestFile>,
189        mut writer: ManifestWriter,
190    ) -> Result<ManifestFile> {
191        for manifest_file in manifest_bin {
192            let manifest_file = manifest_file.load_manifest(&file_io).await?;
193            for manifest_entry in manifest_file.entries() {
194                if manifest_entry.status() == ManifestStatus::Deleted
195                    && manifest_entry
196                        .snapshot_id()
197                        .is_some_and(|id| id == snapshot_id)
198                {
199                    //only files deleted by this snapshot should be added to the new manifest
200                    writer.add_delete_entry(manifest_entry.as_ref().clone())?;
201                } else if manifest_entry.status() == ManifestStatus::Added
202                    && manifest_entry
203                        .snapshot_id()
204                        .is_some_and(|id| id == snapshot_id)
205                {
206                    //added entries from this snapshot are still added, otherwise they should be existing
207                    writer.add_entry(manifest_entry.as_ref().clone())?;
208                } else if manifest_entry.status() != ManifestStatus::Deleted {
209                    // add all non-deleted files from the old manifest as existing files
210                    writer.add_existing_entry(manifest_entry.as_ref().clone())?;
211                }
212            }
213        }
214
215        writer.write_manifest_file().await
216    }
217
218    async fn merge_group(
219        &self,
220        snapshot_produce: &mut SnapshotProducer<'_>,
221        first_manifest: &ManifestFile,
222        group_manifests: Vec<ManifestFile>,
223    ) -> Result<Vec<ManifestFile>> {
224        let packer: ListPacker<ManifestFile> = ListPacker::new(self.target_size_bytes);
225        let manifest_bins =
226            packer.pack(group_manifests, |manifest| manifest.manifest_length as u32);
227
228        let manifest_merge_futures = manifest_bins
229            .map(|manifest_bin| {
230                if manifest_bin.len() == 1 {
231                    Ok(Box::pin(async { Ok(manifest_bin) })
232                        as Pin<
233                            Box<dyn Future<Output = Result<Vec<ManifestFile>>> + Send>,
234                        >)
235                }
236                //  if the bin has the first manifest (the new data files or an appended manifest file) then only
237                //  merge it if the number of manifests is above the minimum count. this is applied only to bins
238                //  with an in-memory manifest so that large manifests don't prevent merging older groups.
239                else if manifest_bin
240                    .iter()
241                    .any(|manifest| manifest == first_manifest)
242                    && manifest_bin.len() < self.min_count_to_merge as usize
243                {
244                    Ok(Box::pin(async { Ok(manifest_bin) })
245                        as Pin<
246                            Box<dyn Future<Output = Result<Vec<ManifestFile>>> + Send>,
247                        >)
248                } else {
249                    let writer = snapshot_produce.new_manifest_writer(self.content)?;
250                    let snapshot_id = snapshot_produce.snapshot_id;
251                    let file_io = snapshot_produce.table.file_io().clone();
252                    Ok((Box::pin(async move {
253                        Ok(vec![
254                            self.merge_bin(
255                                snapshot_id,
256                                file_io,
257                                manifest_bin,
258                                writer,
259                            )
260                            .await?,
261                        ])
262                    }))
263                        as Pin<Box<dyn Future<Output = Result<Vec<ManifestFile>>> + Send>>)
264                }
265            })
266            .collect::<Result<Vec<Pin<Box<dyn Future<Output = Result<Vec<ManifestFile>>> + Send>>>>>()?;
267
268        let merged_bins: Vec<Vec<ManifestFile>> =
269            futures::future::join_all(manifest_merge_futures.into_iter())
270                .await
271                .into_iter()
272                .collect::<Result<Vec<_>>>()?;
273
274        Ok(merged_bins.into_iter().flatten().collect())
275    }
276
277    async fn merge_manifest(
278        &self,
279        snapshot_produce: &mut SnapshotProducer<'_>,
280        manifests: Vec<ManifestFile>,
281    ) -> Result<Vec<ManifestFile>> {
282        if manifests.is_empty() {
283            return Ok(manifests);
284        }
285
286        let first_manifest = manifests[0].clone();
287
288        let group_manifests = self.group_by_spec(manifests);
289
290        let mut merge_manifests = vec![];
291        for (_spec_id, manifests) in group_manifests.into_iter().rev() {
292            merge_manifests.extend(
293                self.merge_group(snapshot_produce, &first_manifest, manifests)
294                    .await?,
295            );
296        }
297
298        Ok(merge_manifests)
299    }
300}

302struct MergeManifsetProcess {
303    target_size_bytes: u32,
304    min_count_to_merge: u32,
305}
306
307impl ManifestProcess for MergeManifsetProcess {
308    async fn process_manifests(
309        &self,
310        snapshot_produce: &mut SnapshotProducer<'_>,
311        manifests: Vec<ManifestFile>,
312    ) -> Result<Vec<ManifestFile>> {
313        let (unmerg_data_manifests, unmerge_delete_manifest) = manifests
314            .into_iter()
315            .partition(|m| m.content == ManifestContentType::Data);
316        let mut data_manifests = {
317            let merge_manifest_manager = MergeManifestManager::new(
318                self.target_size_bytes,
319                self.min_count_to_merge,
320                ManifestContentType::Data,
321            );
322            merge_manifest_manager
323                .merge_manifest(snapshot_produce, unmerg_data_manifests)
324                .await?
325        };
326        data_manifests.extend(unmerge_delete_manifest);
327        Ok(data_manifests)
328    }
